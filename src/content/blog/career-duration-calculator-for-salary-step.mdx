## 네이버 계산기 쓰다가 호봉 틀리는 이유

직장에서 경력증명서 발급 업무 해본 사람들은 알 거야. 경력 날짜 계산하는 게 진짜 여간 힘든 일이 아니거든.

급하게 **네이버 날짜 계산기** 켜서 일수 계산하는 경우가 있는데, 이거 절대 쓰면 안 돼! 일반 날짜 계산기는 윤년이나 월별 일수 차이, 그리고 호봉 산정 특유의 **'월/일 환산 기준'**을 정확히 반영하지 못하거든. 잘못 산정했다가 나중에 급여 호봉 꼬이면 진짜 대참사 일어난다구.

## 기존 엑셀 계산기의 뼈아픈 단점들

보통 급여 담당자들 사이에서 돌아다니는 standard 엑셀 서식을 많이들 쓰잖아? 나도 매번 그거 썼거든. 근데 쓸 때마다 솔직히 너무 불편했어.

수식 깨질까 봐 덜덜 떠는 것도 싫고, 입력 칸 잘못 건드렸다가 결과값 꼬이는 일도 수두룩했지. 무엇보다 외근 나가거나 다른 PC에서 급하게 계산해야 할 때 엑셀 파일 찾아서 열고 하는 과정 자체가 엄청난 번거로움이었어.

## 직접 만든 자동 경력기간 산출 프로그램!

그래서 답답한 사람이 우물 판다고, 내가 직접 웹이랑 PC 양쪽에서 모두 쓸 수 있는 **경력 자동 계산 프로그램**을 만들어버렸어!

입력창에 입사일이랑 퇴사일만 쏙 쏙 넣어주면, **호봉 산정 기준에 딱 맞춰서 총 경력기간이 연, 월, 일 단위로 자동 산출**돼. 실수할 확률은 0%에 수렴하고, 작업 시간은 10분의 1로 줄어들었지.

{/* 3. 포스트 전용 1회용 JavaScript (MDX 이벤트 리스너 바인딩) */}
<script>{`
  document.addEventListener("DOMContentLoaded", function () {
    // 1. 날짜 입력 자동 하이픈(-) 바인딩
    const startDateInput = document.getElementById('wp-start-date');
    const endDateInput = document.getElementById('wp-end-date');
    const calcBtn = document.getElementById('btn-calc-career');

    function autoHyphen(target) {
      let value = target.value.replace(/[^0-9]/g, '');
      if (value.length > 4 && value.length <= 6) {
        target.value = value.slice(0, 4) + '-' + value.slice(4);
      } else if (value.length > 6) {
        target.value = value.slice(0, 4) + '-' + value.slice(4, 6) + '-' + value.slice(6, 8);
      } else {
        target.value = value;
      }
    }

    if (startDateInput) {
      startDateInput.addEventListener('input', (e) => autoHyphen(e.target));
    }
    if (endDateInput) {
      endDateInput.addEventListener('input', (e) => autoHyphen(e.target));
    }

    // 2. 경력 계산 함수
    if (calcBtn) {
      calcBtn.addEventListener('click', function () {
        const startVal = startDateInput.value;
        const endVal = endDateInput.value;

        if (startVal.length < 10 || endVal.length < 10) {
          alert('시작일과 종료일을 정확히 입력해주세요. (예: 20260421)');
          return;
        }

        const start = new Date(startVal);
        const end = new Date(endVal);

        if (isNaN(start.getTime()) || isNaN(end.getTime())) {
          alert('유효한 날짜 형식이 아닙니다.');
          return;
        }

        if (start > end) {
          alert('종료일은 시작일보다 이후여야 합니다.');
          return;
        }

        // 시작일과 종료일 당일 포함 (+1일)
        let endAdjusted = new Date(end);
        endAdjusted.setDate(endAdjusted.getDate() + 1);

        let years = endAdjusted.getFullYear() - start.getFullYear();
        let months = endAdjusted.getMonth() - start.getMonth();
        let days = endAdjusted.getDate() - start.getDate();

        if (days < 0) {
          months -= 1;
          const startMonthLastDay = new Date(start.getFullYear(), start.getMonth() + 1, 0).getDate();
          days += startMonthLastDay;
        }

        if (months < 0) {
          years -= 1;
          months += 12;
        }

        const diffTime = Math.abs(end - start);
        const totalDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24)) + 1;

        document.getElementById('wp-result-text').innerText = \`\${years}년 \${months}개월 \${days}일\`;
        document.getElementById('wp-result-days').innerText = totalDays.toLocaleString();
        document.getElementById('wp-result-box').style.display = 'block';
      });
    }

    // 3. 페이드인 애니메이션
    const fadeElements = document.querySelectorAll(".sk-fade-in");
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add("visible");
          observer.unobserve(entry.target);
        }
      });
    }, { threshold: 0.15 });

    fadeElements.forEach(el => observer.observe(el));
  });
`}</script>
