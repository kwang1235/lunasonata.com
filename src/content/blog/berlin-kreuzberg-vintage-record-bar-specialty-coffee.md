(() => {
  const initInteractive = () => {
    const fadeElements = document.querySelectorAll('.sk-fade');
    if (fadeElements.length) {
      const observer = new IntersectionObserver((entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            entry.target.style.opacity = '1';
            entry.target.style.transform = 'translateY(0)';

            const bars = entry.target.querySelectorAll('.sk-bar');
            bars.forEach((bar) => {
              const targetWidth = bar.getAttribute('data-width');
              if (targetWidth) bar.style.width = targetWidth;
            });

            const counters = entry.target.querySelectorAll('.sk-counter');
            counters.forEach((counter) => {
              const targetNum = parseInt(counter.getAttribute('data-target') || '0', 10);
              let currentNum = 0;
              const duration = 1000;
              const stepTime = 16;
              const increment = targetNum / (duration / stepTime);

              const updateCounter = () => {
                currentNum += increment;
                if (currentNum < targetNum) {
                  counter.textContent = Math.ceil(currentNum) + '%';
                  setTimeout(updateCounter, stepTime);
                } else {
                  counter.textContent = targetNum + '%';
                }
              };
              updateCounter();
            });

            observer.unobserve(entry.target);
          }
        });
      }, { threshold: 0.15 });

      fadeElements.forEach((el) => {
        el.style.opacity = '0';
        el.style.transform = 'translateY(16px)';
        el.style.transition = 'opacity 0.6s cubic-bezier(0.16, 1, 0.3, 1), transform 0.6s cubic-bezier(0.16, 1, 0.3, 1)';
        observer.observe(el);
      });
    }
  };

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initInteractive);
  } else {
    initInteractive();
  }
  document.addEventListener('astro:page-load', initInteractive);
})();
