<script>
  (function () {
    // 스크롤 감지 Intersection Observer 설정
    const observerOptions = {
      root: null,
      rootMargin: "0px 0px -40px 0px",
      threshold: 0.12
    };

    const fadeObserver = new IntersectionObserver((entries, observer) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add("is-visible");
          observer.unobserve(entry.target);
        }
      });
    }, observerOptions);

    const fadeElements = document.querySelectorAll(".blog-post-content .fade-up");
    fadeElements.forEach(el => fadeObserver.observe(el));

    // 인용구 클릭 인터랙션
    const quotes = document.querySelectorAll(".blog-post-content .post-quote");
    quotes.forEach(quote => {

        quote.style.transition = "transform 0.15s ease, background-color 0.15s ease";

          quote.style.transform = "scale(1)";
          quote.style.backgroundColor = "#f1f5f9";

      });
    });
