  const initInteractive = () => {
    // 1. 공통 스크롤 페이드 인터랙션
    const elements = document.querySelectorAll('.sk-fade');
    if (elements.length) {
      const fadeObserver = new IntersectionObserver((entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            entry.target.style.opacity = '1';
            entry.target.style.transform = 'translateY(0)';
            fadeObserver.unobserve(entry.target);
          }
        });
      }, { threshold: 0.1 });

      elements.forEach((el) => {
        el.style.opacity = '0';
        el.style.transform = 'translateY(16px)';
        el.style.transition = 'opacity 0.6s cubic-bezier(0.16, 1, 0.3, 1), transform 0.6s cubic-bezier(0.16, 1, 0.3, 1)';
        fadeObserver.observe(el);
      });
    }

    // 2. KT 위약금 카운터 인터랙션
    const penaltyWrap = document.getElementById('sk-kt-penalty-display');
    if (penaltyWrap && !penaltyWrap.dataset.observed) {
      penaltyWrap.dataset.observed = 'true';
      const penaltyTargets = [
        { id: 'sk-penalty-val-1', target: 141681, duration: 2000 },
        { id: 'sk-penalty-val-2', target: 73025, duration: 1850 }
      ];
      let hasPenaltyExecuted = false;

      const easeOutQuint = (x) => 1 - Math.pow(1 - x, 5);

      const animateGentleCount = (el, targetValue, duration) => {
        const start = performance.now();
        const frame = (now) => {
          const elapsed = now - start;
          const progress = Math.min(elapsed / duration, 1);
          const easedProgress = easeOutQuint(progress);
          const currentVal = Math.round(targetValue * easedProgress);
          el.innerText = currentVal.toLocaleString();
          if (progress < 1) {
            requestAnimationFrame(frame);
          } else {
            el.innerText = targetValue.toLocaleString();
          }
        };
        requestAnimationFrame(frame);
      };

      const penaltyObserver = new IntersectionObserver((entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting && !hasPenaltyExecuted) {
            hasPenaltyExecuted = true;
            penaltyTargets.forEach((item) => {
              const el = document.getElementById(item.id);
              if (el) animateGentleCount(el, item.target, item.duration);
            });
            penaltyObserver.unobserve(penaltyWrap);
          }
        });
      }, { threshold: 0.35 });

      penaltyObserver.observe(penaltyWrap);
    }

    // 3. 입금 시뮬레이터 로직
    const depositTarget = document.getElementById('sk-deposit-simulator-wrapper');
    if (depositTarget && !depositTarget.dataset.observed) {
      depositTarget.dataset.observed = 'true';
      const depositItems = [

      ];

      let isProcessing = false;
      let hasAutoStarted = false;
      let audioContext = null;

      const playDepositChime = () => {
        try {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          if (!AudioCtx) return;
          if (!audioContext) audioContext = new AudioCtx();
          if (audioContext.state === 'suspended') audioContext.resume();
          const now = audioContext.currentTime;
          const osc = audioContext.createOscillator();
          const gain = audioContext.createGain();
          osc.type = 'sine';
          osc.frequency.setValueAtTime(987.77, now);
          osc.frequency.setValueAtTime(1318.51, now + 0.08);
          gain.gain.setValueAtTime(0.18, now);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.55);
          osc.connect(gain);
          gain.connect(audioContext.destination);
          osc.start(now);
          osc.stop(now + 0.55);
        } catch (e) {}
      };

      const runDepositFlow = () => {
        if (isProcessing) return;
        isProcessing = true;

        const badge = document.getElementById('sk-status-badge');
        const balanceText = document.getElementById('sk-current-balance');
        const listContainer = document.getElementById('sk-history-list');

        const counterText = document.getElementById('sk-counter-text');

        if (!balanceText || !listContainer) return;

            setTimeout(() => {

            }, 1200);

              badge.style.backgroundColor = '#059669';
            }

            isProcessing = false;
            return;
          }

          const cur = depositItems[step];
          playDepositChime();

          balanceText.innerText = cur.balance.toLocaleString();
          if (counterText) counterText.innerText = (step + 1) + '건 완료';

          if (alertBox && alertTime && alertMsg) {
            alertBox.style.display = 'block';
            alertBox.style.opacity = '1';
            alertTime.innerText = cur.time;
            alertMsg.innerHTML = '<span style="color: #0b50a2;">' + cur.sender + '</span>님으로부터 <span style="color: #0284c7;">' + cur.amount.toLocaleString() + '원</span>이 입금되었습니다.';

          card.className = 'sk-item-card';
          card.style.cssText = 'display: flex; justify-content: space-between; align-items: center; padding: 12px 14px; background-color: #f8fafc; border-radius: 12px; border: 1px solid #e2e8f0; box-sizing: border-box;';
          card.innerHTML = `
            <div style="min-width: 0;">

          listContainer.insertBefore(card, listContainer.firstChild);

          step++;
          setTimeout(runStep, 800);
        };

        runStep();
      };

      const depositObserver = new IntersectionObserver((entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting && !hasAutoStarted) {
            hasAutoStarted = true;
            runDepositFlow();
            depositObserver.unobserve(depositTarget);
          }
        });
      }, { threshold: 0.35 });

      depositObserver.observe(depositTarget);
    }
  };

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initInteractive);
  } else {
    initInteractive();
  }
  document.addEventListener('astro:page-load', initInteractive);
})();
