function toggleSkAccordion(btn) {
  const item = btn.closest('.sk-accordion-item');
  if (!item) return;
  const content = item.querySelector('.sk-accordion-content');
  const chevron = btn.querySelector('.sk-chevron-icon');
  
  if (content.style.maxHeight && content.style.maxHeight !== '0px') {
    content.style.maxHeight = '0px';
    if (chevron) chevron.style.transform = 'rotate(0deg)';
  } else {
    content.style.maxHeight = content.scrollHeight + 'px';
    if (chevron) chevron.style.transform = 'rotate(180deg)';
  }
}

function switchSkOverseasTab(type) {
  const btnActual = document.getElementById('sk-os-tab-btn-actual');
  const btnFixed = document.getElementById('sk-os-tab-btn-fixed');
  const panelActual = document.getElementById('sk-os-panel-actual');
  const panelFixed = document.getElementById('sk-os-panel-fixed');

  if (!btnActual || !btnFixed || !panelActual || !panelFixed) return;

  if (type === 'actual') {
    btnActual.style.background = '#0d9488';
    btnActual.style.color = '#ffffff';
    btnActual.style.borderColor = '#0d9488';

    btnFixed.style.background = '#f8fafc';
    btnFixed.style.color = '#475569';
    btnFixed.style.borderColor = '#cbd5e1';

    panelActual.style.display = 'block';
    panelFixed.style.display = 'none';
  } else {
    btnFixed.style.background = '#0d9488';
    btnFixed.style.color = '#ffffff';
    btnFixed.style.borderColor = '#0d9488';

    btnActual.style.background = '#f8fafc';
    btnActual.style.color = '#475569';
    btnActual.style.borderColor = '#cbd5e1';

    panelFixed.style.display = 'block';
    panelActual.style.display = 'none';
  }
}

(() => {
  const initInteractive = () => {
    const elements = document.querySelectorAll('.sk-fade');
    if (elements.length) {
      const observer = new IntersectionObserver((entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            entry.target.style.opacity = '1';
            entry.target.style.transform = 'translateY(0)';
            observer.unobserve(entry.target);
          }
        });
      }, { threshold: 0.1 });

      elements.forEach((el) => {
        el.style.opacity = '0';
        el.style.transform = 'translateY(16px)';
        el.style.transition = 'opacity 0.6s cubic-bezier(0.16, 1, 0.3, 1), transform 0.6s cubic-bezier(0.16, 1, 0.3, 1)';
        observer.observe(el);
      });
    }

    const gradeEl = document.getElementById('sk-calc-grade');
    const nightsEl = document.getElementById('sk-calc-nights');
    const fxEl = document.getElementById('sk-calc-fx');
    const krwFxEl = document.getElementById('sk-calc-krw-fx');
    const titleEl = document.getElementById('sk-calc-overseas-title');
    const descEl = document.getElementById('sk-calc-overseas-desc');

    if (gradeEl && nightsEl && fxEl && krwFxEl && titleEl && descEl) {
      const calculateOverseas = () => {
        const usdRate = parseFloat(gradeEl.value) || 137;
        const nights = parseInt(nightsEl.value, 10) || 1;
        const fxRate = parseFloat(fxEl.value) || 1.0;
        const krwFx = parseFloat(krwFxEl.value) || 1350;

        const totalUsdLimit = usdRate * nights;
        const localLimit = Math.round(totalUsdLimit * fxRate);
        const discountUsd = Math.round(usdRate * 0.85) * nights;
        const krwLimit = Math.floor((totalUsdLimit * krwFx) / 10) * 10;

        titleEl.innerHTML = `${localLimit.toLocaleString('ko-KR')} (현지화) <span style="font-size: 0.88rem; font-weight: 600; color: #0f766e;">/ 85% 정액 ${discountUsd}$</span>`;
        descEl.innerText = `총 상한 ${totalUsdLimit}$ (약 ${krwLimit.toLocaleString('ko-KR')}원) | 1박당 ${usdRate}$ × ${nights}박 기준 환산 결과입니다.`;
      };

      gradeEl.addEventListener('change', calculateOverseas);
      nightsEl.addEventListener('input', calculateOverseas);
      fxEl.addEventListener('input', calculateOverseas);
      krwFxEl.addEventListener('input', calculateOverseas);
    }
  };

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initInteractive);
  } else {
    initInteractive();
  }
  document.addEventListener('astro:page-load', initInteractive);
})();
